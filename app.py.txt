@st.cache_data(ttl=3600)
def get_vie_comune(comune):
    try:
        q = (
            '[out:json][timeout:10];'
            f'area[name="{comune}"][admin_level=8]->.searchArea;'
            '(way(area.searchArea)["highway"]["name"];);'
            'out 100;'
        )
        url = "https://overpass-api.de/api/interpreter"
        r = requests.post(url, data={"data": q}, timeout=8)
        if r.status_code == 200:
            data = r.json()
            vie = sorted(list(set([
                el["tags"]["name"]
                for el in data.get("elements", [])
                if "tags" in el and "name" in el["tags"]
            ])))
            if vie:
                return ["-- Seleziona Via --"] + vie[:200]
        return [
            "-- Seleziona Via --",
            "Via Roma","Via Garibaldi","Via Milano",
            "Via Verdi","Via Dante","Via Sacco",
            "Corso Matteotti","Piazza Liberta"
        ]
    except:
        return ["-- Seleziona Via --","Via Roma","Via Sacco"]
